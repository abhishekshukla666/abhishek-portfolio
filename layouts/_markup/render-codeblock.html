{{- $lang := .Type | default "text" -}} {{- $label := .Attributes.title | default (cond (in (slice "text" "plaintext" "") $lang) "" $lang) -}} {{- $html := transform.Highlight .Inner $lang (merge .Options (dict "noClasses" false)) | string -}} {{- /* Chroma's Swift lexer marks most names as plain `n`. Tell them apart the way Xcode does: @attributes, method calls, .properties and Capitalised type names each get their own class. */ -}} {{- if eq $lang "swift" -}} {{- /* Keywords newer than Chroma's Swift lexer: concurrency, ownership, generics. */ -}} {{- $html = replaceRE `(actor|async|await|some|any|nonisolated|isolated|sending|consuming|borrowing|consume|discard|macro|package|each|repeat)` `$1` $html -}} {{- $html = replaceRE `@(\w+)` `@$1` $html -}} {{- $html = replaceRE `([a-z_]\w*)(\()` `$1$2` $html -}} {{- $html = replaceRE `([A-Z]\w*)` `$1` $html -}} {{- $html = replaceRE `(\.)([a-z_]\w*)` `$1$2` $html -}} {{- end -}}
{{ $label }}
{{- $html | safeHTML -}}
